5. Uncertainty Weighting

\begin{align}
    % 1. Stability Loss (Log-Tikhonov)
    \mathcal{L}_{\text{stab}} &= \frac{1}{N} \sum_{i=1}^{N} \ln \left( \frac{\| \mathbf{r}_i - \mathbf{A}\mathbf{e}_i \|_2^2 + \lambda \| \mathbf{e}_i \|_2^2}{\| \mathbf{r}_i \|_2^2} + \epsilon \right) \\
    % 2. Precision Loss (Log-Residual)
    \mathcal{L}_{\text{prec}} &= \frac{1}{N} \sum_{i=1}^{N} \ln \left( \frac{\| \mathbf{r}_i - \mathbf{A}\mathbf{e}_i \|_2^2}{\| \mathbf{r}_i \|_2^2} + \epsilon \right) \\
    % 3. Total Weighted Loss (Homoscedastic Uncertainty)
    \mathcal{L}_{\text{total}} &= \frac{1}{2} e^{-s_1} \mathcal{L}_{\text{stab}} + \frac{1}{2} s_1 + \frac{1}{2} e^{-s_2} \mathcal{L}_{\text{prec}} + \frac{1}{2} s_2
\end{align}

\noindent \textbf{Where:}
\begin{itemize}
    \item $\lambda = 10^{-4}$ (Tikhonov regularization coefficient)
    \item $\epsilon = 10^{-16}$ (Numerical stability floor)
    \item $s_1, s_2$ are learnable parameters representing the log-variance ($\log \sigma^2$) of the task uncertainty.
\end{itemize}
